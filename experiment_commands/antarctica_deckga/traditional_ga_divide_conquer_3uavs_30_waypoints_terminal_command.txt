import numpy as np
import pickle
from pathlib import Path

txt_in = Path("data/points/antarctica_aspa135_m3_30_points.txt")
pkl_out = Path("data/points/antarctica_aspa135_m3_30_points.pkl")

pts = np.genfromtxt(txt_in, dtype=float, names=True, delimiter=None)
xyz = np.vstack([pts["x"], pts["y"], pts["z"]]).T

mn = xyz.min(axis=0)
mx = xyz.max(axis=0)

print("Loaded:", txt_in.resolve())
print("Shape :", xyz.shape)
print(f"Ranges: x[{mn[0]:.2f},{mx[0]:.2f}] y[{mn[1]:.2f},{mx[1]:.2f}] z[{mn[2]:.2f},{mx[2]:.2f}]")

if xyz.shape != (30, 3):
    raise SystemExit(f"STOP: Expected 30 waypoints, got {xyz.shape[0]}.")

with pkl_out.open("wb") as f:
    pickle.dump(xyz, f)

print("Saved:", pkl_out.resolve())
PY

Terminal 3B — Check transformed flight values
cd ~/Documents/GitHub/ROS2_MultiUAV_3D

python3 - << 'PY'
import pickle
import numpy as np

p = "deckga_ros2/data/traditional_ga_divide_conquer_output_antarctica_30_uav3.pkl"

zin_min = 32.09
zin_max = 39.98
zout_min = 3.80
zout_max = 6.80
z_base = 32.2

def map_z(z):
    return z_base + (z - zin_min) / (zin_max - zin_min) * (zout_max - zout_min) + zout_min

with open(p, "rb") as f:
    d = pickle.load(f)

if "deckga_paths" in d:
    paths = d["deckga_paths"]
elif "paths" in d:
    paths = d["paths"]
else:
    print("Available keys:", d.keys())
    raise SystemExit("STOP: Could not find path key.")

print("Transform check:")
print(f"Input z range  : [{zin_min:.2f}, {zin_max:.2f}]")
print(f"Output z offset: [{zout_min:.2f}, {zout_max:.2f}]")
print(f"z_base         : {z_base:.2f}")
print(f"Final flight z : [{z_base + zout_min:.2f}, {z_base + zout_max:.2f}]")
print(f"Takeoff target : {(z_base + zout_min + 0.10):.2f} using takeoff_margin=0.10")

for i, P in enumerate(paths):
    P = np.asarray(P, dtype=float)
    Q = P.copy()
    Q[:,2] = map_z(P[:,2])

    print(f"\n=== TRANSFORMED UAV{i} flight path ===")
    print(f"Number of points including start/end: {len(Q)}")
    print(f"x range: [{Q[:,0].min():.2f}, {Q[:,0].max():.2f}]")
    print(f"y range: [{Q[:,1].min():.2f}, {Q[:,1].max():.2f}]")
    print(f"z range: [{Q[:,2].min():.2f}, {Q[:,2].max():.2f}]")
    for k, pt in enumerate(Q, start=1):
        print(f"{k:02d}: {pt[0]:8.2f}\t{pt[1]:8.2f}\t{pt[2]:8.2f}")
PY
