Terminal 1 — Convert TXT to PKL
cd ~/Documents/GitHub/ROS2_MultiUAV_3D

ls -la data/points/antarctica_aspa135_m3_60_points.txt
head -n 5 data/points/antarctica_aspa135_m3_60_points.txt

python3 - << 'PY'
import numpy as np
import pickle
from pathlib import Path

txt_in = Path("data/points/antarctica_aspa135_m3_60_points.txt")
pkl_out = Path("data/points/antarctica_aspa135_m3_60_points.pkl")

pts = np.genfromtxt(txt_in, dtype=float, names=True, delimiter=None)
xyz = np.vstack([pts["x"], pts["y"], pts["z"]]).T

print("Loaded:", txt_in.resolve())
print("Shape :", xyz.shape)

mn = xyz.min(axis=0)
mx = xyz.max(axis=0)
print(f"Ranges: x[{mn[0]:.2f},{mx[0]:.2f}] y[{mn[1]:.2f},{mx[1]:.2f}] z[{mn[2]:.2f},{mx[2]:.2f}]")

pkl_out.parent.mkdir(parents=True, exist_ok=True)
with pkl_out.open("wb") as f:
    pickle.dump(xyz, f)

print("Saved:", pkl_out.resolve())
PY
